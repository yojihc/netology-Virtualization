```sql

user@thinkpc:~$ sudo apt update && sudo apt install mysql-client-core-8.0

user@thinkpc:~$ mysql -V

user@thinkpc:~$ mysql -h 127.0.0.1 -P 3306 -u root -p secret

mysql> status

mysql> CREATE USER 'sys_temp'@'%' IDENTIFIED BY 'secret';

mysql> SELECT user, host FROM mysql.user;

mysql> SHOW GRANTS FOR 'sys_temp'

user@thinkpc:~$ mysql -h 127.0.0.1 -P 3306 -u sys_temp -p
```
